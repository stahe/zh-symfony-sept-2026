    2. 动作、路由、响应

    本章考察请求所经过的路径，从浏览器到处理它的动作，再到该动作返回的响应。视图要到第 4 章才会讲到：这里的动作直接返回文本或 JSON。

    • [bin/console]：[Symfony] 的命令行工具；
    • [composer.json]：[Composer] 的项目描述文件。

    <?php
// public/index.php：所有 HTTP 请求的入口点

use App\Kernel;
use Symfony\Component\Dotenv\Dotenv;
use Symfony\Component\ErrorHandler\Debug;
use Symfony\Component\HttpFoundation\Request;

// Composer 的类加载器（examples/vendor 文件夹，共享）
// 本示例的类（App\ 命名空间）位于其 src/ 文件夹中
$loader = require dirname(__DIR__, 2).'/vendor/autoload.php';
$loader->addPsr4('App\\', dirname(__DIR__).'/src/');

// 配置：.env 文件中的变量
(new Dotenv())->bootEnv(dirname(__DIR__).'/.env');

// 在调试模式下，任何 PHP 错误都会变成异常，并显示详细信息
if ($_SERVER['APP_DEBUG']) {
    Debug::enable();
}

// 应用程序的内核处理当前请求，并返回响应
$kernel = new Kernel($_SERVER['APP_ENV'], (bool) $_SERVER['APP_DEBUG']);
$request = Request::createFromGlobals();
$response = $kernel->handle($request);
$response->send();
$kernel->terminate($request, $response);
    • 第 1 行：[PHP] 文件以 [<?php] 标签开头。其后的一切都是 [PHP] 代码；只包含代码的文件不会关闭这个标签；
    • 第 2 行：注释，从 [//] 到行尾（[/* ... */] 界定多行注释）；
    • 第 4–7 行：语句 [use] 导入一个类名。[PHP] 中类的完整名称包含其命名空间 （namespace），它相当于 [Java] 中的包：[Symfony\Component\HttpFoundation\Request] 是命名空间 [Request] 中的类 [Symfony\Component\HttpFoundation]。在 [use] 之后，文件中可以直接写 [Request]。[Symfony] 中类的名称说明了它们的来源：[Component\HttpFoundation]，表示请求的组件；HTTP，表示响应的组件；[Component\Dotenv]，读取文件 [.env] 的组件；[Component\ErrorHandler]，处理错误的组件。[App\Kernel] 是应用自身的一个类；
    • 第 11 行：[require] 执行另一个文件，[PHP]。[__DIR__] 是当前文件所在的目录（这里是 [public]）；[dirname(__DIR__, 2)] 向上两级到达 [examples] 文件夹；[.] 运算符 连接 两个字符串。文件 [vendor/autoload.php] 由 [Composer] 写入，它安装类加载器：第一次使用某个类时，它会找到并加载定义该类的文件。多亏了这一点，你无需再为类编写 [require] 。[vendor] 文件夹属于 [examples] 文件夹，由所有示例共享（见简介）。[require] 会返回被执行文件的值：[vendor/autoload.php] 返回加载器，存放在变量 [$loader] 中。在 [PHP] 中，变量名总是以 [$] 开头；
    • 第 12 行：[Composer] 的类加载器知道位于 [vendor] 文件夹中各个库的类。它还必须知道应用中的类，按惯例这些类存放在应用的 [src] 文件夹中，位于 [App] 命名空间下。在标准项目中，[Composer] 从项目的 [composer.json] 文件获取这一信息。我们的示例共享同一个名为 [vendor] 的文件夹，但每个示例都有自己的 [src] 文件夹：因此每个示例都通过加载器的 [addPsr4] 方法自行向它声明（[PSR-4] 标准把命名空间映射到文件夹：[App\Controller\FirstController] 类位于 [src/Controller/FirstController.php] 文件中）。运算符 [->] 表示访问对象的成员。在用单引号括起的字符串中，[\\] 表示单个字符 [\]；
    • 第 15 行：[new Dotenv()] 创建一个 [Dotenv] 类的对象，其方法 [bootEnv] 读取文件 [.env]（下文介绍），并把其中的变量存入 [$_SERVER] 和 [$_ENV] 这两个 [PHP] 数组中，应用将在那里找到它们；
    • 第 18–20 行：调试模式 （文件 [.env] 根据环境推断出它；见下文）。[Debug::enable()] 会把任何 [PHP] 错误——即使只是一个简单的警告——转换为异常，并显示在详细的错误页面上（示例 03）。[::] 表示访问类的成员：[enable] 是一个静态方法，可以不通过对象而直接在类上调用；
    • 第 23 行：创建应用的内核，即 [App\Kernel] 类的一个对象，带有两个参数：环境（[dev]）和调试模式（[true]）。[(bool)] 把值转换为布尔值；
    • 第 24 行：[Request::createFromGlobals()] 创建一个 [Request] 对象，内容取自 [PHP] 收到的 HTTP 请求（[PHP] 把它存放在它的“超全局”数组 [$_GET]、[$_POST]、[$_COOKIE]、[$_SERVER] 等中）；
    • 第 25 行：内核处理请求（[handle]）并返回一个 [Response] 对象。一切都发生在这里：内核启动（组装其 Bundle 并构建其服务），找到负责该请求的动作，执行它，并取回它产生的响应；
    • 第 26 行：响应被发送给浏览器：先是状态码、头部，然后是正文；
    • 第 27 行：内核结束请求：它执行那些可以等到响应发送之后再处理的任务（见生命周期一章）。

    这四行概括了 [Symfony]：请求进，响应出。整个应用就是一个把 [Request] 对象转换为 [Response] 对象的函数。这个文件对所有示例都是相同的。它还突出了 [PHP] 的一个独特之处：应用在每个请求时都会重新构建。Web 服务器对它收到的每个请求都会从头到尾执行 [public/index.php] ：加载类、构建内核、处理请求、发送响应，然后忘掉一切。没有任何对象会从一个请求存活到下一个请求。在数据作用域一章中，我们会看到这种行为的后果，它与 [Java]、[C#] 或 [Python] 恰恰相反，后者的应用在请求之间一直驻留在内存中。

    如果每次都必须重新读取并解释全部配置，那么为每个请求都构建内核的代价会很高。[Symfony] 只做一次这件事：它把配置编译成 [PHP] 代码，存放在 [var/cache/dev] 文件夹中（针对 [dev] 环境），然后重复使用。在调试模式下，它会在每个请求时检查配置文件是否已更改，必要时重新编译。

    典型的 [Symfony] 项目有一个更短的 [public/index.php] 文件，它把这项任务委托给 [Runtime] 组件（来自 [Symfony] ，案例研究就使用它）：我们选择完整地写出来，以便看清发生了什么。

    <?php
// src/Kernel.php：应用程序的内核，最小的 Symfony 应用程序

namespace App;

use Symfony\Bundle\FrameworkBundle\Kernel\MicroKernelTrait;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\HttpKernel\Kernel as BaseKernel;
use Symfony\Component\Routing\Attribute\Route;

class Kernel extends BaseKernel
{
    // 内核的标准行为：config/bundles.php 中的 Bundle、
    // config/packages 的配置、路由
    use MicroKernelTrait;

    // 一个操作（ACTION）：处理 GET / 请求时调用的方法。#[Route] 属性
    // 将它与 / 路径关联起来
    #[Route('/')]
    public function home(): Response
    {
        // 发送给浏览器的响应
        return new Response('Hello from Symfony!');
    }
}
    • 第 4 行：类的命名空间。因此它的完整名称是 [App\Kernel]；
    • 第 11 行：应用的 [Kernel] 类继承（[extends]）自 [Kernel] 类，该类来自 [Symfony]，并以名称 [BaseKernel] 导入（第 8 行的关键字 [as] 用于重命名导入，因为这两个类同名）。这个类负责处理请求（[handle]）；
    • 第 15 行：语句 [use] 在类内部并不导入名称；而是引入trait，即一个可复用的类片段。[MicroKernelTrait] 为内核提供其标准行为：它读取 [config/bundles.php] 中的 Bundle 列表、[config/packages] 文件夹中的 Bundle 配置、[config/services.yaml] 中的服务（示例 06），以及 [config/routes.yaml] 中的路由（示例 02）。它还会在 [Kernel] 类自身中查找路由：这使得第一个示例可以只由它构成；
    • 第 19–24 行：一个动作，即处理请求的 public 方法。属性 [#[Route('/')]] 把它与路径 [/] 关联起来：这就是路由。[PHP] 属性写作 [#[...]]，是附加在代码（类、方法、参数）上的信息，[PHP] 会忽略它，但库会读取它：相当于 [Java] 的注解或 [Python] 的装饰器。[Symfony] 大量使用它们；
    • 第 20 行：[: Response] 声明返回值的类型。在 [PHP] 中类型是可选的，但一旦指定了类型，[PHP] 就会在运行时检查它们；
    • 第 23 行：动作返回一个 [Response] 对象，其第一个参数是响应的正文。这遵循 [Symfony] 的规则：动作总是返回一个 [Response] 对象（示例 03）。

    [Symfony] 由Bundle组成：即把功能集成到应用中的包（[Twig] 视图、安全、数据库访问等）。应用的 Bundle 列表位于文件 [config/bundles.php] 中：

// config/bundles.php：构成应用程序的 Bundle（构建块），
// 以及每个 Bundle 处于活动状态的环境
return [
    Symfony\Bundle\FrameworkBundle\FrameworkBundle::class => ['all' => true],

    该文件返回一个关联数组：对每个以类名标识的 Bundle（[FrameworkBundle::class] 是字符串形式的完整类名），它列出了该 Bundle 在哪些环境中处于激活状态（这里是所有环境）。我们的应用只有一个 Bundle，即 FrameworkBundle，它是框架的核心，负责组装基本组件：请求和响应、内核、路由、服务容器、控制台等。后续示例会添加更多 Bundle。每个 Bundle 都有自己的配置，位于 [config/packages] 文件夹中，每个 Bundle 一个文件，用 [YAML] 编写：

    • 第 5 行：应用的密钥：一个随机字符串，[Symfony] 用它对某些数据（登录链接、防 CSRF 令牌等）进行签名。每个示例都有自己的密钥。

    与其他框架不同，[Symfony] 项目的 [.env] 文件通常会与源代码一起发布：它包含变量的默认值，适合开发使用。特定于某次安装的值（例如密码或生产环境密钥）存放在 [.env.local] 文件中，该文件在主文件之后读取，但绝不会发布；在生产环境中，这些值改由真正的环境变量提供，环境变量的优先级高于这两个文件。配置中未指定的内容保持各个 Bundle 所定义的默认值。[php bin/console config:dump-reference framework] 命令会显示 FrameworkBundle 中所有可能的键及其默认值（几百行）。

    #!/usr/bin/env php
<?php
// bin/console：Symfony 的命令行工具（php bin/console debug:router...）

use App\Kernel;
use Symfony\Bundle\FrameworkBundle\Console\Application;
use Symfony\Component\Console\Input\ArgvInput;
use Symfony\Component\Dotenv\Dotenv;

// 类加载器和配置，与 public/index.php 中相同
$loader = require dirname(__DIR__, 2).'/vendor/autoload.php';
$loader->addPsr4('App\\', dirname(__DIR__).'/src/');

(new Application($kernel))->run(new ArgvInput());
    • 第 1 行：在 [Linux] 和 [macOS] 下，这一行使文件可以被直接执行（[./bin/console]）；在 [Windows] 下，则始终输入 [php bin/console]；

    <?php
// src/Controller/FirstController.php：/first/... URL 的操作

namespace App\Controller;

use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\Routing\Attribute\Route;
use Symfony\Component\Routing\Requirement\Requirement;

// 类上的 #[Route] 属性：其各个操作的路由所共享的前缀，
// 既用于路径（/first），也用于名称（first_）
#[Route('/first', name: 'first_')]
class FirstController
{
    // GET /first：'' 路径表示前缀本身
    #[Route('', name: 'index', methods: ['GET'])]
    public function index(): Response
    {
        return new Response('Action [index] of the [FirstController] controller');
    }

    // GET /first/action01
    #[Route('/action01', name: 'action01', methods: ['GET'])]
    public function action01(): Response
    {
        return new Response('Action [action01] of the [FirstController] controller, GET request');
    }

    // POST /first/action01：相同的路径，不同的 HTTP 方法，不同的操作
    #[Route('/action01', name: 'action01_post', methods: ['POST'])]
    public function action01Post(): Response
    {
        return new Response('Action [action01Post] of the [FirstController] controller, POST request');
    }

    // GET /first/products/17：{id} 是路由参数，传递给操作的 $id
    // 参数；requirement 只接受数字，int 类型
    // 将其转换为整数
    #[Route('/products/{id}', name: 'product', requirements: ['id' => '\d+'], methods: ['GET'])]
    public function product(int $id): Response
    {
        return new Response("Product no. $id");
    }

    // GET /first/products/17/reviews/3：两个路由参数；requirement
    // 直接写在路径中，位于 < 和 > 之间
    #[Route('/products/{id<\d+>}/reviews/{number<\d+>}', name: 'review', methods: ['GET'])]
    public function review(int $id, int $number): Response
    {
        return new Response("Review no. $number of product no. $id");
    }

    // GET /first/files/images/logo.png：Requirement::CATCH_ALL（.+
    // 表达式）接受任何路径结尾，包括斜杠
    #[Route('/files/{path}', name: 'file', requirements: ['path' => Requirement::CATCH_ALL],
        methods: ['GET'])]
    public function file(string $path): Response
    {
        return new Response("Requested file: $path");
    }
}
    • 第 10–13 行：应用于[#[Route]] 类的 属性 为其所有动作的路由提供了前缀 ：[/first] 用于路径，[first_] 用于名称。动作的 URL 是该前缀与其路由路径的拼接；
    • 第 16–20 行：路径 [''] 指的是前缀本身：[GET /first]；
    • 第 23–34 行：同一路径，对应两种 HTTP 方法：[GET] 和 [POST]，由两个不同的动作处理。动作的名称可以自由选择：把它与 URL 和 HTTP 方法关联起来的是属性 [#[Route]] 。[methods: ['GET', 'POST']] 则会把两种方法关联到同一个动作；
    • 第 39–43 行：[{id}] 是一个路由参数：URL 的这一段可以取多个值。动作的 [$id] 参数会接收它的值：对于 URL [/first/products/17]，它是 17。默认情况下，路由参数接受任何不含斜杠的文本；[requirements] 参数对它进行限制：正则表达式 [\d+] （一个或多个数字，其中 [\d] 表示一个数字，[+] 表示重复）只接受数字。否则，路由不匹配。动作参数的类型 [int] 指示 [Symfony] 把该值（从 URL 中提取的字符串）转换为整数。这就是数据绑定 ，下一章会详细讲解。在用双引号括起的字符串 ["Product no. $id"]中，变量 [$id] 会被替换为它的值：这就是插值 （相反，用单引号括起的字符串则按原样书写）；
    • 第 47–51 行：两个路由参数被传递给两个动作参数。[Symfony] 按名称映射它们；动作参数的顺序无关紧要。这一次，约束直接写在路径本身中，位于 [<] 和 [>] 之间：这样更简短；
    • 第 55–60 行：一个包含 [/] 的路由参数。[Requirement] 类（来自 [Symfony] ）以常量的形式汇集了最常用的正则表达式：[Requirement::CATCH_ALL] （[.+]：任意字符，包括斜杠）、[Requirement::DIGITS] （[[0-9]+]）、[Requirement::UUID]、[Requirement::DATE_YMD]……

    <?php
// src/Controller/ResponsesController.php：操作的各种响应

namespace App\Controller;

use App\Model\Person;
use RuntimeException;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\Routing\Attribute\Route;

// AbstractController：控制器的父类，为它们提供
// 快捷方法（$this->json、$this->redirectToRoute、$this->render...）
#[Route('/responses', name: 'responses_', methods: ['GET'])]
class ResponsesController extends AbstractController
{
    // 一个字符串，声明为纯文本：浏览器按原样显示
    #[Route('/text', name: 'text')]
    public function text(): Response
    {
        return new Response('<h1>Action [text]</h1>', Response::HTTP_OK,
            ['Content-Type' => 'text/plain; charset=utf-8']);
    }

    // 未声明类型时，Symfony 以 text/html 类型发送响应
    #[Route('/html', name: 'html')]
    public function html(): Response
    {
        return new Response('<h1>Action [html]</h1>');
    }

    // 一个对象：$this->json 将其序列化为 JSON，类型为 [application/json]。
    // 该操作不能命名为 json：那样会重新定义继承的方法
    #[Route('/json', name: 'json')]
    public function asJson(): JsonResponse
    {
        return $this->json(new Person('Martin', 'Jean', 42));
    }

    // 一个关联数组：它会变成 JSON 对象
    #[Route('/array', name: 'array')]
    public function array(): JsonResponse
    {
        return $this->json(['lastName' => 'Martin', 'firstName' => 'Jean', 'age' => 42]);
    }

    // 一个索引数组：它会变成 JSON 数组
    #[Route('/list', name: 'list')]
    public function list(): JsonResponse
    {
        return $this->json(['Mme', 'Melle', 'Mr']);
    }

    // 一个计算得到的数组：先排序，然后逐项转换
    #[Route('/calc', name: 'calc')]
    public function calc(): JsonResponse
    {
        $numbers = [3, 1, 2];
        sort($numbers);
        return $this->json(array_map(fn (int $n) => $n * 10, $numbers));
    }

    // 使用 201 状态码（Created）而不是 200（OK）
    #[Route('/created', name: 'created')]
    public function created(): JsonResponse
    {
        return $this->json(['message' => 'resource created'], Response::HTTP_CREATED);
    }

    // 没有响应：204 状态码（No Content）表示“没有内容可显示”
    #[Route('/empty', name: 'empty')]
    public function empty(): Response
    {
        return new Response(null, Response::HTTP_NO_CONTENT);
    }

    // 一个“预期的”错误：带有 404 状态码的 JSON 响应
    #[Route('/not-found', name: 'not_found')]
    public function notFound(): JsonResponse
    {
        return $this->json(['message' => 'This resource does not exist'], Response::HTTP_NOT_FOUND);
    }

    // 同样的效果，通过 HTTP 异常实现，Symfony 会将其转换为 404 响应
    #[Route('/abort', name: 'abort')]
    public function abort(): never
    {
        throw $this->createNotFoundException('This resource does not exist either');

    为了处理一个请求，[Symfony] 内核要经过一系列步骤，并在每一步触发一个事件：请求到达时触发 [kernel.request] ，动作被选定时触发 [kernel.controller] ，响应生成时触发 [kernel.response] ，依此类推。（生命周期一章会全面介绍它们。）监听器 （listener）是一个类，[Symfony] 会为给定类型的每个事件调用它；它可以查看和修改事件携带的数据。[Symfony] 本身就是这样构造的：路由、安全和会话都是监听器。

    • 第 13 行：属性 [#[AsEventListener]] 使该类成为事件 [kernel.response] （常量 [KernelEvents::RESPONSE]）的监听器。无需再声明其他任何东西：该类是一个服务（示例 02），并且 [Symfony] 会读取它的属性（[autoconfigure]）；
    • 第 16 行：声明在类上的监听器通过其方法 [__invoke] 被调用。它接收事件，即一个 [ResponseEvent] 对象，其中带有请求和响应；
    • 第 18–24 行：如果响应是 JSON 响应（运算符 [instanceof] 用于检查对象的类），监听器就会修改其编码选项：非 ASCII 字符和斜杠按原样写出，并采用缩进格式。运算符 [|] 把这三个选项组合起来，它们是 [PHP] 的常量；[setEncodingOptions] 会重新编码响应数据。

    示例 05 至 12 复用了这个监听器。命令 [php bin/console debug:event-dispatcher kernel.response] 会列出事件监听器：我们的监听器就在 [Symfony] 自带的那些之中。

// src/Products/InMemoryProductsService.php：该契约的一个实现，使用内存中的数据

namespace App\Products;

class InMemoryProductsService implements ProductsService
{
    /** @var list<Product> */
    private array $products;

    public function __construct()
    {
        $this->products = [
            new Product(1, 'pen', 150),
            new Product(2, 'notebook', 320),
            new Product(3, 'eraser', 80),
        ];
    }

    public function findAll(): array
    {
        // PHP 数组在传递时会被复制：调用方收到的是一个副本
        return $this->products;
    }

    public function findById(int $id): ?Product
    {
        // array_find：函数返回 true 的第一个元素，或者 null
        return array_find($this->products, fn (Product $p) => $p->id === $id);
    }
}

    {

        // 价格之和，以分为单位：整数，因此计算是精确的
        $total = array_reduce($products, fn (int $sum, Product $p) => $sum + $p->priceCents, 0);
